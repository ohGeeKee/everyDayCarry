Configure BGP
C8KV-ENT-PEER2# conf t
C8KV-ENT-PEER2(config)# router bgp 65534
C8KV-ENT-PEER2(config-router)# neighbor 192.168.255.6 remote-as 65534
C8KV-ENT-PEER2(config-router)# address-family ipv4
C8KV-ENT-PEER2(config-router-af)# neighbor 192.168.255.6 activate
C8KV-ENT-PEER2(config-router-af)# neighbor 192.168.255.6 next-hop-self
! ensures all routes have the local router address set as the next hop to avoid IBGP routing failure

Steer Outgoing Traffic

C8KV-ENT-CORE# show ip route
! route to 10.9.0.0 via 192.168.255.1
C8KV-ENT-CORE# show ip bgp 10.9.0.0/24 best-path-reason

C8KV-ENT-PEER2# conf t
! intermediate router on the preferred path
C8KV-ENT-PEER2(config)# route-map SET-LOCAL-PREF permit 10
C8KV-ENT-PEER2(config-route-map)# set local-preference 200
! default preference is 100
! higher prreference takes precedence
C8KV-ENT-PEER2(config-route-map)# exit
C8KV-ENT-PEER2(config)# router bgp 65534
C8KV-ENT-PEER2(config-router)# address-family ipv4
C8KV-ENT-PEER2(config-router-af)# neighbor 172.16.2.1 route-map SET-LOCAL-PREF in
! route-map is applied to all received routes for neighbor 172.16.2.1(ISP2)
C8KV-ENT-PEER2(config-router-af)# end
C8KV-ENT-PEER2# show ip route
! verify changes
! route to 10.9.0.0 via 192.168.255.5
C8KV-ENT-CORE# show ip bgp 10.9.0.0/24 best-path-reason

++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++

Steer Incoming Traffic

C8KV-ENT-ISP9# show ip bgp 192.168.0.0 best-path-reason
! route to 192.168.0.0 via 172.16.31.1
! best path is currently AS 64512 65534
C8KV-ENT-PEER1(config)# route-map SET-COMMUNITY permit 10
! sets the 64512:104 community on the edge router for the AS 65534
C8KV-ENT-PEER1(config-route-map)# set community 64512:104 additive
! additive adds the community without overwriting the others
C8KV-ENT-PEER1(config-route-map)# router bgp 65534
C8KV-ENT-PEER1(config-route-map)# address-family ipv4
C8KV-ENT-PEER1(config-route-map-af)# neighbor 172.16.1.1 route-map SET-COMMUNITY out
! route-map is attached to the outgoing direction to the ISP1 neighbor(AS 64512)
C8KV-ENT-PEER1(config-route-map-af)# neighbor 172.16.1.1 send-community

C8KV-ISP1# show ip bgp 192.168.0.0
! verify send-community worked
! Community: [64512:104]
C8KV-ISP1# show ip bgp 192.168.0.0 best-path-reason
